Calculate Risk Scores¶

Risk Score Formula: Risk Score = Impact × Likelihood

Risk Levels:

  • Critical: Score ≥ 15
  • High: Score 10-14
  • Medium: Score 5-9
  • Low: Score < 5

Recall: risk levels are Critical (≥15), High (10-14), Medium (5-9), Low (<5)

def calculate_risk_score(df):
    """
    Calculate risk scores for all risks in the dataframe.
    
    Args:
        df: DataFrame with 'impact' and 'likelihood' columns
    
    Returns:
        DataFrame with 'risk_score' column added
    """
    # Step 1: Calculate risk_score = df['impact'] * df['likelihood']
    # Step 2: Return the updated dataframe
    df['risk_score'] = df['impact'] * df['likelihood']
    
    return df

# Step 2: Assign risk_level based on score thresholds
def assign_risk_level(score):
    """
    Assign risk level based on risk score.
    
    Expected output format:
    - score >= 15: 'Critical'
    - 10 <= score < 15: 'High'
    - 5 <= score < 10: 'Medium'
    - score < 5: 'Low'
    """
    # Implement conditional logic to assign levels
    if score >= 15:
        return 'Critical'
    elif score >= 10:
        return 'High'
    elif score >= 5:
        return 'Medium'
    else:
        return 'Low'
    pass

# Ranking function signature
def rank_risks_by_score(df):
    """
    Sort risks by risk_score in descending order and add a rank column.
    """
    # Step 1: Sort by risk_score descending
    df_sorted = df.sort_values(by='risk_score', ascending=False)
    
    # Step 2: Reset index
    df_sorted = df_sorted.reset_index(drop=True)
    
    # Step 3: Add 'rank' column starting from 1
    df_sorted['rank'] = df_sorted.index + 1
    
    # Step 4: Return sorted dataframe
    return df_sorted

# Function to print distribution statistics
def print_risk_statistics(df):
    """
    Print risk distribution statistics by level, category, and NIST function.
    """
    total = len(df)
    
    # Step 1: Count risks by 'risk_level' and print with percentages
    print("=" * 50)
    print("RISK LEVEL DISTRIBUTION")
    print("=" * 50)
    level_counts = df['risk_level'].value_counts()
    for level, count in level_counts.items():
        pct = (count / total) * 100
        print(f"  {level:<10}: {count:>2} risks ({pct:.1f}%)")
    
    # Step 2: Count risks by 'category' and print with percentages
    print("\n" + "=" * 50)
    print("RISK BY CATEGORY")
    print("=" * 50)
    category_counts = df['category'].value_counts()
    for category, count in category_counts.items():
        pct = (count / total) * 100
        print(f"  {category:<15}: {count:>2} risks ({pct:.1f}%)")
    
    # Step 3: Count risks by 'nist_function' and print with percentages
    print("\n" + "=" * 50)
    print("RISK BY NIST FUNCTION")
    print("=" * 50)
    function_counts = df['nist_function'].value_counts()
    for function, count in function_counts.items():
        pct = (count / total) * 100
        print(f"  {function:<10}: {count:>2} risks ({pct:.1f}%)")
    
    # Step 4: Calculate and print mean, median, max, min of 'risk_score'
    print("\n" + "=" * 50)
    print("SCORE STATISTICS")
    print("=" * 50)
    print(f"  Average : {df['risk_score'].mean():.2f}")
    print(f"  Median  : {df['risk_score'].median():.2f}")
    print(f"  Max     : {df['risk_score'].max()}")
    print(f"  Min     : {df['risk_score'].min()}")

import matplotlib.pyplot as plt
plt.close('all')

def create_heat_map(df):
    fig, ax = plt.subplots(figsize=(12, 9))
    color_map = {
        'Technical': '#1f77b4',
        'Ethical': '#d62728',
        'Operational': '#ff7f0e',
        'Legal': '#2ca02c',
    }
    for category in df['category'].unique():
        subset = df[df['category'] == category]
        ax.scatter(subset['impact'], subset['likelihood'],
                   s=300, c=color_map.get(category, 'gray'),
                   label=category, alpha=0.7,
                   edgecolors='black', linewidth=1.5)
    for _, row in df.iterrows():
        ax.annotate(row['id'], (row['impact'], row['likelihood']),
                    fontsize=9, fontweight='bold',
                    ha='center', va='center', color='white')
    ax.axvline(x=3, color='gray', linestyle='--', linewidth=1.5, alpha=0.7)
    ax.axhline(y=3, color='gray', linestyle='--', linewidth=1.5, alpha=0.7)
    ax.set_xlabel('Impact', fontsize=12)
    ax.set_ylabel('Likelihood', fontsize=12)
    ax.set_title('AI Risk Heat Map — FraudShield AI', fontsize=14, fontweight='bold')
    ax.set_xlim(0.5, 5.5)
    ax.set_ylim(0.5, 5.5)
    ax.set_xticks([1, 2, 3, 4, 5])
    ax.set_yticks([1, 2, 3, 4, 5])
    ax.legend(title='Category', loc='upper left')
    ax.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.show()

import matplotlib.pyplot as plt
import numpy as np
plt.close('all')

def create_category_bar_chart(df):
    stats = df.groupby('category')['risk_score'].agg(['min', 'mean', 'max'])
    stats = stats.sort_index()
    
    categories = stats.index.tolist()
    mins = stats['min'].values
    means = stats['mean'].values
    maxs = stats['max'].values
    
    x = np.arange(len(categories))
    width = 0.25
    
    fig, ax = plt.subplots(figsize=(12, 7))
    
    bars_min = ax.bar(x - width, mins, width, label='Minimum', color='#2ca02c', edgecolor='black')
    bars_mean = ax.bar(x, means, width, label='Average', color='#1f77b4', edgecolor='black')
    bars_max = ax.bar(x + width, maxs, width, label='Maximum', color='#d62728', edgecolor='black')
    
    for bars in [bars_min, bars_mean, bars_max]:
        for bar in bars:
            height = bar.get_height()
            ax.annotate(f'{height:.1f}',
                        xy=(bar.get_x() + bar.get_width() / 2, height),
                        xytext=(0, 3),
                        textcoords='offset points',
                        ha='center', va='bottom',
                        fontsize=9, fontweight='bold')
    
    ax.set_xlabel('Risk Category', fontsize=12)
    ax.set_ylabel('Risk Score (Impact × Likelihood)', fontsize=12)
    ax.set_title('Risk Score Distribution by Category — FraudShield AI', fontsize=14, fontweight='bold')
    ax.set_xticks(x)
    ax.set_xticklabels(categories, fontsize=11)
    ax.set_ylim(0, max(maxs) + 5)
    ax.legend(title='Statistic', loc='upper right', fontsize=10)
    ax.grid(True, axis='y', alpha=0.3, linestyle='--')
    ax.set_axisbelow(True)
    
    plt.tight_layout()
    plt.show()

def generate_executive_summary(df, df_ranked):
    """
    Generate and print a formatted executive summary report.
    """
    total = len(df)
    
    # Step 1: Print header with total risk count
    print("=" * 70)
    print("EXECUTIVE SUMMARY — AI RISK ASSESSMENT")
    print("FraudShield AI | NIST AI RMF Aligned")
    print("=" * 70)
    print(f"\nTotal Risks Identified: {total}\n")
    
    # Step 2: Count and print each risk level with percentages
    print("-" * 70)
    print("RISK LEVEL BREAKDOWN")
    print("-" * 70)
    level_order = ['Critical', 'High', 'Medium', 'Low']
    level_counts = df['risk_level'].value_counts().reindex(level_order, fill_value=0)
    for level in level_order:
        count = level_counts[level]
        pct = (count / total) * 100
        print(f"  {level:<10}: {count:>2} risks ({pct:>5.1f}%)")
    
    # Step 3: Calculate and print average risk score
    avg_score = df['risk_score'].mean()
    max_score = df['risk_score'].max()
    min_score = df['risk_score'].min()
    print("\n" + "-" * 70)
    print("SCORE STATISTICS")
    print("-" * 70)
    print(f"  Average Risk Score : {avg_score:.2f}")
    print(f"  Maximum Risk Score : {max_score}")
    print(f"  Minimum Risk Score : {min_score}")
    
    # Step 4: Get top 5 risks from df_ranked
    top5 = df_ranked.head(5)
    
    # Step 5: Print top 5 risks with formatted details
    print("\n" + "-" * 70)
    print("TOP 5 PRIORITY RISKS")
    print("-" * 70)
    for _, row in top5.iterrows():
        # Truncate long descriptions to keep the report readable
        desc = row['description']
        if len(desc) > 70:
            desc = desc[:67] + "..."
        print(f"\n  Rank {row['rank']} | {row['id']} | Score: {row['risk_score']} ({row['risk_level']})")
        print(f"  {desc}")
    
    # Step 6: Print conclusion statement about mitigation priorities
    critical_count = level_counts['Critical']
    critical_pct = (critical_count / total) * 100
    
    print("\n" + "=" * 70)
    print("CONCLUSION & RECOMMENDED ACTION")
    print("=" * 70)
    if critical_pct >= 50:
        priority_msg = (
            f"With {critical_pct:.0f}% of risks classified as Critical, FraudShield AI "
            f"exhibits a systemic risk posture that requires immediate executive attention. "
            f"Deployment should be paused pending remediation of the {critical_count} Critical risks, "
            f"starting with {top5.iloc[0]['id']} (score {top5.iloc[0]['risk_score']})."
        )
    elif critical_pct >= 25:
        priority_msg = (
            f"With {critical_pct:.0f}% of risks classified as Critical, FraudShield AI "
            f"requires prioritized remediation. The {critical_count} Critical risks should be "
            f"addressed before any model updates or scope expansion."
        )
    else:
        priority_msg = (
            f"Risk posture is manageable, with only {critical_pct:.0f}% of risks in the Critical tier. "
            f"Continue monitoring and address the {critical_count} Critical risks within the next sprint."
        )
    
    print(f"\n{priority_msg}\n")
    print("=" * 70)
